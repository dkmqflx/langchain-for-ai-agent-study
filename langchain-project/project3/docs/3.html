러너 v1 — 명령 한 번에 점수표

🟢 뼈대 ⏱ 2~3시간
🎯 목표

uv run python runner.py --target rag --dataset rag_v1 --label rerank 한 줄로 시험지 전체를 돌려, 케이스별 점수표(results.csv)와 요약(summary.json)이 폴더 하나에 남게 합니다. 지표는 아직 코드로 셀 수 있는 것만 — 비용 0원, 항상 같은 답이 나오는 것들입니다.

🧠 개념 — 딱 하나만

실행(run) 하나 = 시험지 × 시험 대상 × 설정, 그리고 폴더 하나

P2 STEP 8의 evaluate.py는 결과를 CSV 한 장에 덮어썼습니다. 그걸로는 "지난주와 비교"가 안 돼요. 지난주 파일이 없으니까요. P3는 실행마다 새 폴더를 만듭니다. 이름에 시험지 · 라벨 · 시각이 들어가서, 폴더 목록이 곧 실험 일지가 됩니다.

폴더 안에는 두 파일이 있습니다. results.csv는 케이스별 원본(답변 · 근거 · 지표), summary.json은 평균과 설정입니다. STEP 6의 비교는 summary.json끼리, "어느 케이스가 갈렸나"는 results.csv끼리 봅니다.

실험실 노트는 어제 페이지 위에 오늘 결과를 덧쓰지 않습니다. 날짜와 조건을 적고 새 페이지에 씁니다. 그래야 "지난주 rerank가 오늘 baseline보다 나았나"를 두 페이지를 펼쳐 놓고 비교할 수 있죠. runs/rag_v1__rerank__20260826-143012/가 그 페이지 한 장입니다.

①

metrics.py — 코드로 세는 지표

— True/False를 1.0/0.0으로 두는 이유
metrics.py
import math

from schemas import AgentCase, RagCase, RunResult


def _b(x: bool) -> float:
    return 1.0 if x else 0.0


def score_rag(case: RagCase, r: RunResult) -> dict:
    """P1 응답 하나를 채점. RAGAS 전에도 잴 수 있는 것부터 잰다."""
    if case.expect_source:
        source_hit = _b(any(case.expect_source in s for s in r.sources))
    else:
        source_hit = math.nan                      # 기대 문서를 안 적은 케이스는 '해당 없음' — 평균에서 빠진다
    return {
        "ok": _b(r.error is None),                 # 서비스가 응답했나
        "has_sources": _b(bool(r.sources)),        # 근거를 하나라도 달았나
        "source_hit": source_hit,                  # 기대한 문서가 근거에 들어 있나
        "latency_ms": r.latency_ms,
    }


def score_agent(case: AgentCase, r: RunResult) -> dict:
    """P2 응답 하나를 채점. P2 STEP 8의 세 가지 판정을 그대로 가져온다."""
    used = r.used_tools
    tool_ok = (case.expect_tool in used) if case.expect_tool else (len(used) == 0)
    sources_ok = bool(r.sources) if case.expect_sources else True
    block_ok = r.blocked == case.expect_blocked
    return {
        "ok": _b(r.error is None),
        "tool_ok": _b(tool_ok),                    # 기대한 도구를 썼나 (기대가 없으면 '안 썼나')
        "format_ok": _b(sources_ok and block_ok),  # 근거를 달았고, 막을 걸 막았나
        "block_ok": _b(block_ok),                  # 가드레일만 따로도 본다 — 프롬프트를 고치다 뚫리는 걸 잡기 위해
        "latency_ms": r.latency_ms,
    }


SCORERS = {"rag": score_rag, "agent": score_agent}


def score_case(target: str, case, r: RunResult) -> dict:
    return SCORERS[target](case, r)

지표를 전부 숫자로 두는 게 핵심입니다. True 대신 1.0을 넣으면 15개 케이스의 평균이 곧 "비율"이 되고, pandas의 mean() 한 줄로 요약표가 나옵니다. "해당 없음"은 NaN — 평균에서 자동으로 빠집니다.

runner.py — 인자와 실행 이름

— label은 사람용 이름표, params는 실제 손잡이
runner.py
import argparse
import json
import math
import time
from pathlib import Path

import pandas as pd

from metrics import score_case
from schemas import load_dataset
from targets import call_target, health


def parse_params(pairs: list[str]) -> dict:
    """['mode=rerank', 'k=4'] → {'mode': 'rerank', 'k': '4'}"""
    return dict(p.split("=", 1) for p in pairs)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--target", required=True, choices=["rag", "agent"])
    ap.add_argument("--dataset", required=True)                       # datasets/<name>.json
    ap.add_argument("--label", required=True)                         # 이 실행의 이름표. "rerank", "prompt-B"
    ap.add_argument("--param", nargs="*", default=[])                 # 시험 대상에 넘길 손잡이. mode=baseline
    ap.add_argument("--limit", type=int)                              # 처음엔 3개로 하네스가 도는지부터
    args = ap.parse_args()
    params = parse_params(args.param)

    ds = load_dataset(args.dataset)
    assert ds.target == args.target, f"{ds.name}은 {ds.target}용 데이터셋입니다"
    if not health()[args.target]:
        raise SystemExit(f"❌ {args.target} 서비스가 응답하지 않습니다. 서버를 먼저 띄우세요.")

    cases = ds.cases[: args.limit] if args.limit else ds.cases
    run_id = f"{ds.name}__{args.label}__{time.strftime('%Y%m%d-%H%M%S')}"
    out_dir = Path("runs") / run_id
    out_dir.mkdir(parents=True)
    print(f"▶ {run_id} · {len(cases)}개 · params={params}")

--label과 --param은 다른 것입니다. label은 폴더 이름에 들어가는 사람용 이름표고, param은 시험 대상에 실제로 넘어가는 손잡이예요. 둘 다 summary.json에 남으니 나중에 "이 실행이 어떤 설정이었지"를 잊어버릴 일이 없습니다.

runner.py — 저장과 요약

— 숫자 열의 평균이 곧 요약
    df = pd.DataFrame(rows).set_index("id")
    df.to_csv(out_dir / "results.csv", encoding="utf-8-sig")

    # 요약 = 숫자 열의 평균. True/False를 1/0으로 넣어 뒀으니 평균이 곧 비율이다.
    metrics = df.select_dtypes("number").mean().round(3).to_dict()

               "label": args.label, "params": params, "n": len(df),
               "metrics": {k: (None if math.isnan(v) else v) for k, v in metrics.items()}}
    (out_dir / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")

    print("\n" + "=" * 52)
    print(pd.Series(summary["metrics"]).to_string())
    print("=" * 52)
    print(f"📁 {out_dir}")


if __name__ == "__main__":
    main()

select_dtypes("number")가 이 설계의 보상입니다. STEP 4에서 RAGAS 열 넷, STEP 5에서 judge 열 하나가 붙어도 요약 코드는 한 글자도 안 바뀝니다 — 숫자 열이면 알아서 평균에 들어가니까요. version을 요약에 남기는 건 STEP 6에서 "같은 시험지였나"를 확인하기 위해서입니다.

전체 코드 한 번에 보기 — runner.py (STEP 3 버전) 
# ═══════════════════════════════════════════════════════════════════
# STEP 3 · 러너: 골든셋 × 시험 대상 → 케이스별 점수표 + 요약. 명령 한 번이 실행 하나(run)다.
#
# 실행:  uv run python runner.py --target rag   --dataset rag_v1   --label rerank --param mode=rerank
#        uv run python runner.py --target agent --dataset agent_v1 --label prompt-A
# 결과:  runs/<dataset>__<label>__<시각>/results.csv, summary.json
# ═══════════════════════════════════════════════════════════════════
import argparse
import json
import math
import time
from pathlib import Path

import pandas as pd

from metrics import score_case
from schemas import load_dataset
from targets import call_target, health


def parse_params(pairs: list[str]) -> dict:
    """['mode=rerank', 'k=4'] → {'mode': 'rerank', 'k': '4'}"""
    return dict(p.split("=", 1) for p in pairs)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--target", required=True, choices=["rag", "agent"])
    ap.add_argument("--dataset", required=True)                       # datasets/<name>.json
    ap.add_argument("--label", required=True)                         # 이 실행의 이름표. "rerank", "prompt-B"
    ap.add_argument("--param", nargs="*", default=[])                 # 시험 대상에 넘길 손잡이. mode=baseline
    ap.add_argument("--limit", type=int)                              # 처음엔 3개로 하네스가 도는지부터
    args = ap.parse_args()
    params = parse_params(args.param)

    ds = load_dataset(args.dataset)
    assert ds.target == args.target, f"{ds.name}은 {ds.target}용 데이터셋입니다"
    if not health()[args.target]:
        raise SystemExit(f"❌ {args.target} 서비스가 응답하지 않습니다. 서버를 먼저 띄우세요.")

    cases = ds.cases[: args.limit] if args.limit else ds.cases
    run_id = f"{ds.name}__{args.label}__{time.strftime('%Y%m%d-%H%M%S')}"
    out_dir = Path("runs") / run_id
    out_dir.mkdir(parents=True)
    print(f"▶ {run_id} · {len(cases)}개 · params={params}")

    rows, outputs = [], {}                                            # outputs: id → RunResult (STEP 4에서 contexts를 쓴다)
    for i, case in enumerate(cases, 1):
        result = call_target(ds.target, case, params, run_id)
        scores = score_case(ds.target, case, result)
        row = {"id": case.id, "question": case.question, "answer": result.answer,
               "sources": " / ".join(result.sources), "used_tools": ",".join(result.used_tools),
               "error": result.error, **scores}
        rows.append(row)
        outputs[case.id] = result
        mark = "✅" if scores["ok"] else "❌"
        print(f"  {mark} {i:2d}/{len(cases)} {case.id:10s} {result.latency_ms:6d}ms")

    df = pd.DataFrame(rows).set_index("id")
    df.to_csv(out_dir / "results.csv", encoding="utf-8-sig")

    # 요약 = 숫자 열의 평균. True/False를 1/0으로 넣어 뒀으니 평균이 곧 비율이다.
    metrics = df.select_dtypes("number").mean().round(3).to_dict()
    summary = {"run_id": run_id, "dataset": ds.name, "version": ds.version, "target": ds.target,
               "label": args.label, "params": params, "n": len(df),
               "metrics": {k: (None if math.isnan(v) else v) for k, v in metrics.items()}}
    (out_dir / "summary.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")

    print("\n" + "=" * 52)
    print(pd.Series(summary["metrics"]).to_string())
    print("=" * 52)
    print(f"📁 {out_dir}")


if __name__ == "__main__":
    main()

  • 폴더가 생긴다 — runs/ 아래에 실행마다 폴더가 하나씩, 안에 results.csv·summary.json
  • 요약이 숫자다 — rag는 ok · has_sources · source_hit · latency_ms, agent는 ok · tool_ok · format_ok · block_ok · latency_ms
  • CSV를 읽을 수 있다 — results.csv를 열어 source_hit이 0인 케이스의 sources를 보고 "어느 문서가 대신 나왔는지" 말할 수 있다
  • 시험지가 다르면 거절한다 — --target agent --dataset rag_v1로 실행하면 rag용 데이터셋입니다로 멈춘다
🎓 한 줄 정리

실행 하나는 폴더 하나이고, 지표는 전부 숫자라서 평균이 곧 요약이다. 이 두 규칙 위에 STEP 4~7이 열(column)만 더해 나간다.

더 알아보기 — 왜 평균만 보지 않나 / 같은 설정을 두 번 돌려 보기

평균은 "얼마나", CSV는 "어디서"

source_hit 0.867은 15개 중 2개가 엉뚱한 문서를 근거로 댔다는 뜻입니다. 어느 2개인지는 results.csv에만 있어요. 평균이 떨어졌을 때 할 일은 CSV를 열어 그 케이스의 answer와 sources를 읽는 것입니다. STEP 6이 이 "어디서"를 자동으로 뽑아 줍니다.

같은 설정을 두 번 돌려 보세요

--label rerank를 한 번 더 실행하고 두 summary.json을 나란히 놓아 보세요. LLM은 같은 질문에도 답이 조금씩 달라서 숫자가 흔들립니다. 그 흔들림의 폭이 STEP 6에서 "얼마나 떨어져야 회귀인가"의 임계값을 정하는 근거가 됩니다. 폭보다 작은 변화는 잡음이에요.
